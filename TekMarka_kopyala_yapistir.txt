'=====================================================================
'  TEK MARKA NOZUL PERFORMANSI
'  Makro adı: TekMarkaPerformans
'  1 marka sorar -> o markanın pivotunu oluşturur:
'  Ort. Aldığı Süre, Ort. Aldığı Döküm Sayısı, Kullanım Adedi
'  Filtreler: Marka (tek seçim) ve Değişim Sebebi
'  TekMarkaRapor: Değişim sebebi + toplam süre içeren YENİ Excel raporu
'  NOZUL 2026 sayfası açıkken çalıştırın.
'=====================================================================
Option Explicit

' True : sadece "Değişim Sebebi" girilmiş satırlar alınır
Private Const SADECE_DEGISENLER As Boolean = True

Private mWb As Workbook
Private mN As Long
Private mMarka() As String
Private mSebep() As String
Private mSure() As Double
Private mSureOK() As Boolean
Private mDokum() As Double
Private mDokumOK() As Boolean

Public Sub TekMarkaPerformans()
    Dim dMarka As Object, i As Long, k As String, marka As String
    Dim wsV As Worksheet, wsP As Worksheet
    Dim pc As PivotCache, pt As PivotTable, df As PivotField
    Dim outArr() As Variant, pi As PivotItem

    If Not VeriYukle() Then Exit Sub

    Set dMarka = CreateObject("Scripting.Dictionary")
    For i = 1 To mN
        k = Norm(mMarka(i))
        If Not dMarka.Exists(k) Then dMarka.Add k, mMarka(i)
    Next i
    marka = MarkaSec(dMarka, "Performansına bakılacak marka", "")
    If marka = "" Then Exit Sub

    On Error GoTo Hata
    Application.ScreenUpdating = False
    Set wsV = SayfaHazirla(mWb, "PerfVeri")
    Set wsP = SayfaHazirla(mWb, "MarkaPerformans")

    ReDim outArr(1 To mN + 1, 1 To 5)
    outArr(1, 1) = "Marka"
    outArr(1, 2) = "Değişim Sebebi"
    outArr(1, 3) = "Aldığı Süre"
    outArr(1, 4) = "Aldığı Döküm Sayısı"
    outArr(1, 5) = "Adet"
    For i = 1 To mN
        outArr(i + 1, 1) = dMarka.Item(Norm(mMarka(i)))
        outArr(i + 1, 2) = IIf(mSebep(i) = "", "(Boş)", mSebep(i))
        If mSureOK(i) Then outArr(i + 1, 3) = mSure(i)
        If mDokumOK(i) Then outArr(i + 1, 4) = mDokum(i)
        outArr(i + 1, 5) = 1
    Next i
    wsV.Range("A1").Resize(mN + 1, 5).Value = outArr
    wsV.Range("C2").Resize(mN, 1).NumberFormat = "[h]:mm:ss"

    Set pc = mWb.PivotCaches.Create(SourceType:=xlDatabase, _
                                    SourceData:=wsV.Range("A1").Resize(mN + 1, 5))
    Set pt = pc.CreatePivotTable(TableDestination:=wsP.Range("A8"), TableName:="MarkaPerformans")

    With pt
        .PivotFields("Marka").Orientation = xlPageField
        .PivotFields("Marka").EnableMultiplePageItems = True
        .PivotFields("Değişim Sebebi").Orientation = xlPageField
        .PivotFields("Değişim Sebebi").EnableMultiplePageItems = True

        Set df = .AddDataField(.PivotFields("Aldığı Süre"), "Ort. Aldığı Süre", xlAverage)
        df.NumberFormat = "[h]:mm:ss"
        Set df = .AddDataField(.PivotFields("Aldığı Döküm Sayısı"), "Ort. Aldığı Döküm Sayısı", xlAverage)
        df.NumberFormat = "0.00"
        Set df = .AddDataField(.PivotFields("Adet"), "Kullanım Adedi", xlSum)
        df.NumberFormat = "0"
        .DataPivotField.Orientation = xlColumnField
    End With

    ' Sadece seçilen marka görünsün
    On Error Resume Next
    For Each pi In pt.PivotFields("Marka").PivotItems
        If Norm(pi.Name) <> Norm(marka) Then pi.Visible = False
    Next pi
    Err.Clear
    pt.ManualUpdate = False
    pt.RefreshTable
    On Error GoTo Hata

    With wsP
        .Range("A1").Value = "NOZUL PERFORMANSI"
        .Range("A1").Font.Bold = True
        .Range("A1").Font.Size = 14
        .Range("A2").Value = "Marka değiştirmek için 'Marka' filtresini, sebep seçmek için 'Değişim Sebebi' filtresini kullanın."
        .Range("A2").Font.Italic = True
        .Columns("A:D").ColumnWidth = 26
    End With

    wsV.Visible = xlSheetHidden
    wsP.Activate
    Application.ScreenUpdating = True
    Exit Sub
Hata:
    Application.ScreenUpdating = True
    MsgBox "Hata: " & Err.Description, vbCritical
End Sub

'=====================================================================
'  TekMarkaRapor : 1 marka sorar, YENİ bir Excel dosyası oluşturur.
'  Değişim sebebine göre: Kullanım adedi, %, Toplam aldığı süre,
'  Ort. aldığı süre, Toplam döküm, Ort. döküm. Sebep sütununda filtre var,
'  TOPLAM satırı filtreye göre otomatik değişir.
'=====================================================================
Public Sub TekMarkaRapor()
    Dim dMarka As Object, dSeb As Object
    Dim i As Long, k As String, marka As String, km As String
    Dim nSeb As Long, si As Long, r As Long, adet As Long
    Dim sebAd() As String, sAdet() As Long, sSumS() As Double, sSumD() As Double
    Dim wbR As Workbook, ws As Worksheet, wsD As Worksheet
    Dim rIlk As Long, rSon As Long, rT As Long
    Dim det() As Variant, n As Long

    If Not VeriYukle() Then Exit Sub

    Set dMarka = CreateObject("Scripting.Dictionary")
    For i = 1 To mN
        k = Norm(mMarka(i))
        If Not dMarka.Exists(k) Then dMarka.Add k, mMarka(i)
    Next i
    marka = MarkaSec(dMarka, "Raporu alınacak marka", "")
    If marka = "" Then Exit Sub
    km = Norm(marka)

    ReDim sebAd(1 To mN): ReDim sAdet(1 To mN)
    ReDim sSumS(1 To mN): ReDim sSumD(1 To mN)
    Set dSeb = CreateObject("Scripting.Dictionary")
    For i = 1 To mN
        If Norm(mMarka(i)) = km Then
            adet = adet + 1
            k = Norm(mSebep(i))
            If k = "" Then k = "(BOS)"
            If Not dSeb.Exists(k) Then
                nSeb = nSeb + 1
                dSeb.Add k, nSeb
                If mSebep(i) = "" Then sebAd(nSeb) = "(Boş)" Else sebAd(nSeb) = mSebep(i)
            End If
            si = dSeb.Item(k)
            sAdet(si) = sAdet(si) + 1
            If mSureOK(i) Then sSumS(si) = sSumS(si) + mSure(i)
            If mDokumOK(i) Then sSumD(si) = sSumD(si) + mDokum(i)
        End If
    Next i
    If adet = 0 Then MsgBox "Bu markaya ait kayıt yok.", vbExclamation: Exit Sub

    On Error GoTo Hata
    Application.ScreenUpdating = False
    Set wbR = Workbooks.Add(xlWBATWorksheet)
    Set ws = wbR.Worksheets(1)
    ws.Name = "Rapor"
    ActiveWindow.DisplayGridlines = False

    ws.Range("A1").Value = "NOZUL PERFORMANS RAPORU - " & marka
    ws.Range("A1").Font.Bold = True: ws.Range("A1").Font.Size = 15
    ws.Range("A2").Value = "Kaynak: " & mWb.Name & "   |   " & Format(Now, "dd.mm.yyyy hh:nn") & _
                           "   |   'Değişim Sebebi' başlığındaki filtreden seçim yapabilirsiniz, TOPLAM satırı filtreye göre değişir."
    ws.Range("A2").Font.Italic = True: ws.Range("A2").Font.Color = RGB(89, 89, 89)

    ws.Range("A4:G4").Value = Array("Değişim Sebebi", "Kullanım Adedi", "Oran %", _
        "Toplam Aldığı Süre", "Ort. Aldığı Süre", "Toplam Döküm Sayısı", "Ort. Döküm Sayısı")
    BaslikBicim ws.Range("A4:G4")

    rIlk = 5
    For si = 1 To nSeb
        r = rIlk + si - 1
        ws.Cells(r, 1).Value = sebAd(si)
        ws.Cells(r, 2).Value = sAdet(si)
        ws.Cells(r, 4).Value = sSumS(si)
        ws.Cells(r, 6).Value = sSumD(si)
    Next si
    rSon = rIlk + nSeb - 1
    If nSeb > 1 Then
        ws.Range(ws.Cells(rIlk, 1), ws.Cells(rSon, 7)).Sort _
            Key1:=ws.Cells(rIlk, 2), Order1:=xlDescending, Header:=xlNo
    End If
    For r = rIlk To rSon
        ws.Cells(r, 3).Formula = "=IFERROR(B" & r & "/SUBTOTAL(109,$B$" & rIlk & ":$B$" & rSon & "),0)"
        ws.Cells(r, 5).Formula = "=IF(B" & r & "=0,0,D" & r & "/B" & r & ")"
        ws.Cells(r, 7).Formula = "=IF(B" & r & "=0,0,F" & r & "/B" & r & ")"
    Next r

    rT = rSon + 2
    ws.Cells(rT, 1).Value = "TOPLAM"
    ws.Cells(rT, 2).Formula = "=SUBTOTAL(109,B" & rIlk & ":B" & rSon & ")"
    ws.Cells(rT, 3).Formula = "=IF(B" & rT & "=0,0,1)"
    ws.Cells(rT, 4).Formula = "=SUBTOTAL(109,D" & rIlk & ":D" & rSon & ")"
    ws.Cells(rT, 5).Formula = "=IF(B" & rT & "=0,0,D" & rT & "/B" & rT & ")"
    ws.Cells(rT, 6).Formula = "=SUBTOTAL(109,F" & rIlk & ":F" & rSon & ")"
    ws.Cells(rT, 7).Formula = "=IF(B" & rT & "=0,0,F" & rT & "/B" & rT & ")"
    With ws.Range(ws.Cells(rT, 1), ws.Cells(rT, 7))
        .Font.Bold = True
        .Interior.Color = RGB(255, 242, 204)
        .Borders(xlEdgeTop).LineStyle = xlContinuous
        .Borders(xlEdgeBottom).LineStyle = xlContinuous
    End With

    ws.Range(ws.Cells(rIlk, 3), ws.Cells(rT, 3)).NumberFormat = "0.0%"
    ws.Range(ws.Cells(rIlk, 4), ws.Cells(rT, 5)).NumberFormat = "[h]:mm:ss"
    ws.Range(ws.Cells(rIlk, 6), ws.Cells(rT, 6)).NumberFormat = "0.0"
    ws.Range(ws.Cells(rIlk, 7), ws.Cells(rT, 7)).NumberFormat = "0.00"
    ws.Range(ws.Cells(rIlk, 2), ws.Cells(rT, 7)).HorizontalAlignment = xlCenter
    With ws.Range(ws.Cells(4, 1), ws.Cells(rSon, 7)).Borders
        .LineStyle = xlContinuous
        .Color = RGB(191, 191, 191)
    End With
    ws.Range(ws.Cells(4, 1), ws.Cells(rSon, 7)).AutoFilter
    ws.Columns("A").ColumnWidth = 28
    ws.Columns("B:G").ColumnWidth = 18
    ws.Rows(4).RowHeight = 32

    ' ---- Detay sayfası (markanın tüm kayıtları) ----
    Set wsD = wbR.Worksheets.Add(After:=ws)
    wsD.Name = "Detay"
    ReDim det(1 To adet + 1, 1 To 3)
    det(1, 1) = "Değişim Sebebi": det(1, 2) = "Aldığı Süre": det(1, 3) = "Aldığı Döküm Sayısı"
    n = 1
    For i = 1 To mN
        If Norm(mMarka(i)) = km Then
            n = n + 1
            det(n, 1) = IIf(mSebep(i) = "", "(Boş)", mSebep(i))
            If mSureOK(i) Then det(n, 2) = mSure(i)
            If mDokumOK(i) Then det(n, 3) = mDokum(i)
        End If
    Next i
    wsD.Range("A1").Resize(n, 3).Value = det
    wsD.Range("B2").Resize(n - 1, 1).NumberFormat = "[h]:mm:ss"
    BaslikBicim wsD.Range("A1:C1")
    wsD.Range("A1").Resize(n, 3).AutoFilter
    wsD.Columns("A:C").ColumnWidth = 24

    ws.Activate
    ws.Range("A1").Select
    Application.ScreenUpdating = True
    MsgBox "Rapor yeni bir Excel dosyasında oluşturuldu. Kaydetmeyi unutmayın.", vbInformation
    Exit Sub
Hata:
    Application.ScreenUpdating = True
    MsgBox "Hata: " & Err.Description, vbCritical
End Sub

Private Sub BaslikBicim(rng As Range)
    With rng
        .Font.Bold = True
        .Font.Color = RGB(255, 255, 255)
        .Interior.Color = RGB(31, 56, 100)
        .HorizontalAlignment = xlCenter
        .VerticalAlignment = xlCenter
        .WrapText = True
    End With
End Sub

'=====================================================================
'  YARDIMCI FONKSİYONLAR
'=====================================================================
Private Function VeriYukle() As Boolean
    Dim lo As ListObject
    Dim hdr As Variant, dat As Variant
    Dim cM As Long, cS As Long, cSure As Long, cDok As Long
    Dim r As Long, nR As Long
    Dim mk As String, sb As String, eksik As String

    Set lo = TabloBul()
    If lo Is Nothing Then
        MsgBox "'Marka' ve 'Aldığı Süre' sütunlarını içeren tablo bulunamadı." & vbCrLf & _
               "NOZUL 2026 sayfasını açıp makroyu tekrar çalıştırın.", vbExclamation
        Exit Function
    End If
    If lo.DataBodyRange Is Nothing Then
        MsgBox "Tablo boş.", vbExclamation
        Exit Function
    End If

    Set mWb = lo.Parent.Parent
    hdr = lo.HeaderRowRange.Value2
    dat = lo.DataBodyRange.Value2

    cM = KolonBul(hdr, "MARKA")
    cS = KolonBul(hdr, "DEGISIMSEBEBI")
    cSure = KolonBul(hdr, "ALDIGISURE")
    cDok = KolonBul(hdr, "ALDIGIDOKUMSAYISI")
    If cM = 0 Then eksik = eksik & "Marka, "
    If cS = 0 Then eksik = eksik & "Değişim Sebebi, "
    If cSure = 0 Then eksik = eksik & "Aldığı Süre, "
    If cDok = 0 Then eksik = eksik & "Aldığı Döküm Sayısı, "
    If eksik <> "" Then
        MsgBox "Şu sütun(lar) bulunamadı: " & Left$(eksik, Len(eksik) - 2), vbExclamation
        Exit Function
    End If

    nR = UBound(dat, 1)
    ReDim mMarka(1 To nR): ReDim mSebep(1 To nR)
    ReDim mSure(1 To nR): ReDim mSureOK(1 To nR)
    ReDim mDokum(1 To nR): ReDim mDokumOK(1 To nR)
    mN = 0
    For r = 1 To nR
        mk = Temiz(dat(r, cM))
        sb = Temiz(dat(r, cS))
        If mk <> "" Then
            If (Not SADECE_DEGISENLER) Or sb <> "" Then
                mN = mN + 1
                mMarka(mN) = mk
                mSebep(mN) = sb
                mSure(mN) = SureCevir(dat(r, cSure), mSureOK(mN))
                mDokum(mN) = SayiCevir(dat(r, cDok), mDokumOK(mN))
            End If
        End If
    Next r

    If mN = 0 Then
        MsgBox "Uygun veri satırı bulunamadı.", vbExclamation
        Exit Function
    End If
    VeriYukle = True
End Function

Private Function TabloBul() As ListObject
    Dim ws As Worksheet, lo As ListObject
    On Error Resume Next
    For Each lo In ActiveSheet.ListObjects
        If TabloUygun(lo) Then Set TabloBul = lo: Exit Function
    Next lo
    For Each ws In ActiveWorkbook.Worksheets
        For Each lo In ws.ListObjects
            If TabloUygun(lo) Then Set TabloBul = lo: Exit Function
        Next lo
    Next ws
End Function

Private Function TabloUygun(lo As ListObject) As Boolean
    Dim hdr As Variant
    hdr = lo.HeaderRowRange.Value2
    TabloUygun = (KolonBul(hdr, "MARKA") > 0 And KolonBul(hdr, "ALDIGISURE") > 0)
End Function

Private Function KolonBul(hdr As Variant, anahtar As String) As Long
    Dim j As Long
    For j = LBound(hdr, 2) To UBound(hdr, 2)
        If Norm(hdr(1, j)) = anahtar Then KolonBul = j: Exit Function
    Next j
    For j = LBound(hdr, 2) To UBound(hdr, 2)
        If InStr(1, Norm(hdr(1, j)), anahtar, vbBinaryCompare) > 0 Then KolonBul = j: Exit Function
    Next j
End Function

' Türkçe karakter, boşluk ve satır sonlarını temizleyip büyük harfe çevirir
Private Function Norm(ByVal v As Variant) As String
    Dim s As String
    If IsError(v) Then Exit Function
    s = UCase$(CStr(v))
    s = Replace(s, vbCr, ""): s = Replace(s, vbLf, "")
    s = Replace(s, " ", ""): s = Replace(s, Chr(160), "")
    s = Replace(s, ChrW(775), "")                                  ' birleşik nokta
    s = Replace(s, ChrW(304), "I"): s = Replace(s, ChrW(305), "I") ' İ ı
    s = Replace(s, ChrW(286), "G"): s = Replace(s, ChrW(287), "G") ' Ğ ğ
    s = Replace(s, ChrW(220), "U"): s = Replace(s, ChrW(252), "U") ' Ü ü
    s = Replace(s, ChrW(350), "S"): s = Replace(s, ChrW(351), "S") ' Ş ş
    s = Replace(s, ChrW(214), "O"): s = Replace(s, ChrW(246), "O") ' Ö ö
    s = Replace(s, ChrW(199), "C"): s = Replace(s, ChrW(231), "C") ' Ç ç
    s = Replace(s, "i", "I")
    Norm = s
End Function

Private Function Temiz(ByVal v As Variant) As String
    If IsError(v) Or IsEmpty(v) Then Exit Function
    Temiz = Trim$(Replace(CStr(v), Chr(160), " "))
End Function

' Süre: sayı (Excel saat değeri) veya "03:43:00" gibi metin olabilir
Private Function SureCevir(ByVal v As Variant, ByRef ok As Boolean) As Double
    Dim s As String, p As Variant, h As Double, m As Double, sn As Double, d As Double
    ok = False
    If IsError(v) Or IsEmpty(v) Then Exit Function
    If VarType(v) = vbString Then
        s = Trim$(v)
        If s = "" Then Exit Function
        p = Split(s, ":")
        If UBound(p) < 1 Then Exit Function
        h = Val(p(0)): m = Val(p(1))
        If UBound(p) >= 2 Then sn = Val(Replace(p(2), ",", "."))
        d = (h + m / 60 + sn / 3600) / 24
    ElseIf IsNumeric(v) Then
        d = CDbl(v)
    Else
        Exit Function
    End If
    ok = (d > 0)
    SureCevir = d
End Function

' Döküm sayısı: sayı veya "1,5" gibi metin olabilir
Private Function SayiCevir(ByVal v As Variant, ByRef ok As Boolean) As Double
    Dim s As String, d As Double
    ok = False
    If IsError(v) Or IsEmpty(v) Then Exit Function
    If VarType(v) = vbString Then
        s = Trim$(Replace(v, ",", "."))
        If s = "" Then Exit Function
        If Not (s Like "*#*") Then Exit Function
        d = Val(s)
    ElseIf IsNumeric(v) Then
        d = CDbl(v)
    Else
        Exit Function
    End If
    ok = (d >= 0)
    SayiCevir = d
End Function

Private Function MarkaSec(d As Object, baslik As String, varsayilan As String) As String
    Dim keys As Variant, i As Long, liste As String, defIdx As Long
    Dim cevap As String, kv As String
    keys = d.keys
    kv = Norm(varsayilan)
    For i = 0 To d.Count - 1
        liste = liste & (i + 1) & " - " & d.Item(keys(i)) & vbCrLf
        If defIdx = 0 And keys(i) = kv Then defIdx = i + 1
    Next i
    If defIdx = 0 Then
        For i = 0 To d.Count - 1
            If InStr(keys(i), kv) > 0 Or InStr(kv, keys(i)) > 0 Then defIdx = i + 1: Exit For
        Next i
    End If
    cevap = InputBox(baslik & " numarasını yazın:" & vbCrLf & vbCrLf & liste, _
                     "Marka Seçimi", IIf(defIdx > 0, CStr(defIdx), ""))
    cevap = Trim$(cevap)
    If cevap = "" Then Exit Function
    If IsNumeric(cevap) Then
        i = CLng(cevap)
        If i >= 1 And i <= d.Count Then MarkaSec = d.Item(keys(i - 1)): Exit Function
    End If
    If d.Exists(Norm(cevap)) Then MarkaSec = d.Item(Norm(cevap)): Exit Function
    MsgBox "Geçersiz seçim: " & cevap, vbExclamation
End Function

Private Function SayfaHazirla(wb As Workbook, ad As String) As Worksheet
    Dim ws As Worksheet
    On Error Resume Next
    Set ws = wb.Worksheets(ad)
    On Error GoTo 0
    If Not ws Is Nothing Then
        Application.DisplayAlerts = False
        ws.Visible = xlSheetVisible
        ws.Delete
        Application.DisplayAlerts = True
    End If
    Set ws = wb.Worksheets.Add(After:=wb.Worksheets(wb.Worksheets.Count))
    ws.Name = ad
    Set SayfaHazirla = ws
End Function
